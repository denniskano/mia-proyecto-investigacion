\begin{frame}[t]{3. Cómo se calculan e interpretan}
\vspace{0.15em}
{\small
\begin{itemize}\itemsep0.08em
  \item \textbf{E} = (suma de e\_pair + 0 si no hay par) / ops OFB
  \item \textbf{S} = promedio de similitud de los pares oficiales de E
  \item \textbf{C} = (CR + BQ con par) / (CR + BQ del SD)
  \item \textbf{AlignmentScore} = (S + E + C) / 3
\end{itemize}}
\vspace{0.08em}
{\footnotesize\renewcommand{\arraystretch}{1.08}
\begin{tabularx}{\linewidth}{|L{2.2cm}|L{1.7cm}|Y|}
\hline
\thc{Score} & \thc{Nivel} & \thc{Lectura}\\
\hline
0,80 o más & Alta & Cobertura usable del SD\\
\rowcolor{tablealt}
0,60 a 0,79 & Media & Parcial; declarar tipología\\
0,40 a 0,59 & Baja & Desalineación dominante\\
\rowcolor{tablealt}
menos de 0,40 & Nula & Casi sin correspondencia\\
\hline
\end{tabularx}}
\end{frame}
