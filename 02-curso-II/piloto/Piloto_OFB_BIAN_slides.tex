\begin{frame}[t]{Qué se mide --- AlignmentScore}
\vspace{0.12em}
{\small
\begin{itemize}\itemsep0.32em
  \item \textbf{E} estructural --- greedy 1--1; las ops OFB sin par entran como 0
  \item \textbf{S} semántico --- solo sobre los pares oficiales de E (no pares sueltos)
  \item \textbf{C} cobertura --- (CR + BQ con par) / (CR + BQ del SD)
  \item \textbf{AlignmentScore} = (S + E + C) / 3
\end{itemize}}
\vspace{0.22em}
{\footnotesize\renewcommand{\arraystretch}{1.08}
\begin{tabularx}{\linewidth}{|L{2.2cm}|L{1.7cm}|Y|}
\hline
\thc{Score} & \thc{Nivel} & \thc{Lectura}\\
\hline
$\ge$ 0,80 & Alta & Cobertura usable del SD\\
\rowcolor{tablealt}
0,60 -- 0,79 & Media & Parcial; declarar tipología\\
0,40 -- 0,59 & Baja & Desalineación dominante\\
\rowcolor{tablealt}
$<$ 0,40 & Nula & Casi sin correspondencia\\
\hline
\end{tabularx}}
\end{frame}

\begin{frame}[t]{Qué se concluye}
\vspace{0.2em}
{\small
\begin{enumerate}\itemsep0.55em
  \item El procedimiento S/E/C \textbf{ya corre} sobre un corpus público, versionado y firmado.
  \item Nula en v0 es hallazgo: techo léxico PT$\times$EN. MiniLM lo mueve a Baja \textbf{sin leakage}.
  \item I6 = 0 valida el instrumento: rechaza un par que no comparte recurso.
  \item credit-cards $\times$ CreditCard (0,46 Baja) muestra que el SD existía; faltaba descargarlo.
\end{enumerate}}
\end{frame}
