\begin{frame}{3. Cómo se calculan e interpretan}
Unidad = instancia contrato OFB $\times$ SD BIAN. Oficial: pesos $1/3$.
\vspace{0.35em}

{\small
$E=(\sum e_{\mathrm{pair}}+0\cdot\mathrm{ops\ sin\ par})/n_{\mathrm{OFB}}$\\
$S=$ promedio de similitud de los pares oficiales de $E$\\
$C=(\mathrm{CR+BQ\ con\ par})/(\mathrm{CR+BQ\ del\ SD})$ \quad AlignmentScore $=(S+E+C)/3$
}
\vspace{0.4em}

{\footnotesize
\begin{tabular}{lll}
\rowcolor{tablehead}\tabhead{Score} & \tabhead{Nivel} & \tabhead{Lectura}\\
$\ge 0{,}80$ & Alta & Cobertura usable del SD\\
\rowcolor{tablealt}$0{,}60$--$0{,}79$ & Media & Parcial; declarar tipología\\
$0{,}40$--$0{,}59$ & Baja & Desalineación dominante\\
\rowcolor{tablealt}$< 0{,}40$ & Nula & Casi sin correspondencia\\
\end{tabular}}
\end{frame}
